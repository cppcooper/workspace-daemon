cmake_minimum_required(VERSION 3.20)
project(workspace_daemon VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(workspace_common INTERFACE)
target_include_directories(workspace_common INTERFACE src vendor)
target_compile_options(workspace_common INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(workspace-daemon src/daemon.cpp)
target_link_libraries(workspace-daemon PRIVATE workspace_common Threads::Threads)
add_executable(workspacectl src/client.cpp)
target_link_libraries(workspacectl PRIVATE workspace_common)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py
    --daemon $<TARGET_FILE:workspace-daemon> --client $<TARGET_FILE:workspacectl>)
  set_tests_properties(integration PROPERTIES TIMEOUT 180)
endif()
install(TARGETS workspace-daemon workspacectl RUNTIME DESTINATION bin)
install(PROGRAMS bridge/manager.py DESTINATION libexec/workspace-daemon)
